**Nodo 8 — Parse Enrichment:**
Code node (JavaScript):
```javascript
const lead = $json.lead;
const scrapeResult = $json.scrapeData;
const markdown = scrapeResult?.data?.markdown || '';

const enrichedData = {
  descripcion: (scrapeResult?.data?.metadata?.description || '').substring(0, 500),
  titulo_pagina: (scrapeResult?.data?.metadata?.title || '').substring(0, 200),
  contenido_preview: markdown.substring(0, 1500),
  productos: markdown.match(/(?:productos?|servicios?|soluciones?)\s*[:\-]\s*([^\n]+)/gi)?.slice(0, 5) || [],
  stack_detectado: [],
  tamano_detectado: null,
  industria_detectada: null,
  has_contact_form: markdown.toLowerCase().includes('contacto') || markdown.toLowerCase().includes('contact'),
  has_pricing_page: markdown.toLowerCase().includes('precio') || markdown.toLowerCase().includes('plan'),
  languages: markdown.match(/\b(espanol|english|portugues)\b/gi) || ['espanol'],
  scraped_at: new Date().toISOString(),
  source_enrichment: "firecrawl_scrape",
};

// Detectar stack tecnologico
const stackKeywords = ['hubspot', 'salesforce', 'sap', 'oracle', 'aws', 'azure', 'google cloud', 'shopify', 'magento', 'wordpress', 'pipedrive', 'zoho', 'monday', 'notion', 'slack', 'teams'];
for (const kw of stackKeywords) {
  if (markdown.toLowerCase().includes(kw)) {
    enrichedData.stack_detectado.push(kw);
  }
}

// Detectar tamano por senales en el contenido
if (markdown.match(/\b(1000|mil)\+?\s*(empleados|trabajadores|colaboradores)/i)) {
  enrichedData.tamano_detectado = "1000+";
} else if (markdown.match(/\b(500)\+?\s*(empleados|trabajadores)/i)) {
  enrichedData.tamano_detectado = "500+";
} else if (markdown.match(/\b(100|200|300)\+?\s*(empleados|trabajadores)/i)) {
  enrichedData.tamano_detectado = "100-500";
} else if (markdown.match(/\b(50|60|70|80|90)\+?\s*(empleados|trabajadores)/i)) {
  enrichedData.tamano_detectado = "50-100";
}

// Detectar industria
const industryPatterns = {
  'saas': /\b(saas|software as a service|plataforma|cloud)\b/i,
  'fintech': /\b(fintech|pagos|financier|banking|credito)\b/i,
  'manufactura': /\b(manufactura|fabricacion|industrial|produccion)\b/i,
  'logistica': /\b(logistica|transporte|distribucion|supply chain)\b/i,
  'servicios_profesionales': /\b(consultoria|consulting|asesoria|legal|contabilidad)\b/i,
  'ecommerce': /\b(ecommerce|e-commerce|tienda online|marketplace)\b/i,
  'tecnologia': /\b(tecnologia|technology|desarrollo|software|IT)\b/i,
};
for (const [industry, pattern] of Object.entries(industryPatterns)) {
  if (pattern.test(markdown)) {
    enrichedData.industria_detectada = industry;
    break;
  }
}

return { json: { leadId: lead._id, enrichedData, enrichedBySkill: "leads-enrich-firecrawl" } };
```
