**Nodo 10 — Parse Gemini Response:**
Code node:
```javascript
const response = $json;
const text = response.candidates?.[0]?.content?.parts?.[0]?.text || '{}';
let scoring;
try {
  scoring = JSON.parse(text);
} catch (e) {
  scoring = { score: 0, category: "SKIP", reasoning: "Error parsing Gemini response", signals: [], risks: ["parse_error"], confidence: 0 };
}

// Validar y corregir categoria
const validCategories = ["HOT", "WARM", "NURTURE", "SKIP"];
if (!validCategories.includes(scoring.category)) {
  if (scoring.score >= 80) scoring.category = "HOT";
  else if (scoring.score >= 50) scoring.category = "WARM";
  else if (scoring.score >= 20) scoring.category = "NURTURE";
  else scoring.category = "SKIP";
}

// Costo Gemini Flash Lite (practicamente gratis)
const promptTokens = response.usageMetadata?.promptTokenCount || 0;
const candidateTokens = response.usageMetadata?.candidatesTokenCount || 0;
const costUsd = (promptTokens * 0.075 / 1000000) + (candidateTokens * 0.3 / 1000000);

return {
  json: {
    ...scoring,
    leadId: $json.leadId,
    leadEmpresa: $json.leadEmpresa,
    costUsd: Math.round(costUsd * 10000) / 10000,
    tokensUsed: { input: promptTokens, output: candidateTokens },
  }
};
```
