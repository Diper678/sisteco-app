**En SAAN/convex/leads.ts**, agregar 2 funciones nuevas al final del archivo:

1. **getLeadsToLinkedInEnrich** (query): Retorna leads que:
   - Tienen status "enriched" O "scored" (ya pasaron por enrich basico)
   - Tienen websiteUrl (necesario para derivar URL de LinkedIn)
   - NO tienen datos LinkedIn en enrichedData (enrichedData.linkedin_enriched !== true)

```typescript
export const getLeadsToLinkedInEnrich = query({
  args: { limit: v.optional(v.number()) },
  handler: async (ctx, args) => {
    const leads = await ctx.db
      .query("leads")
      .withIndex("by_status", (q) => q.eq("status", "enriched"))
      .take(args.limit ?? 5);

    // Tambien obtener scored leads que no tienen LinkedIn
    const scoredLeads = await ctx.db
      .query("leads")
      .withIndex("by_status", (q) => q.eq("status", "scored"))
      .take(args.limit ?? 5);

    const all = [...leads, ...scoredLeads];

    // Filtrar: deben tener websiteUrl Y no tener linkedin_enriched
    return all.filter(lead =>
      lead.websiteUrl &&
      (!lead.enrichedData || !lead.enrichedData.linkedin_enriched)
    ).slice(0, args.limit ?? 5);
  },
});
```

2. **enrichLeadLinkedIn** (mutation): Actualiza enrichedData con datos de LinkedIn sin sobreescribir datos existentes:

```typescript
export const enrichLeadLinkedIn = mutation({
  args: {
    leadId: v.id("leads"),
    linkedinData: v.any(), // { companySize, employeeCount, industry, recentPosts, keyContacts, linkedinUrl, linkedin_enriched: true }
  },
  handler: async (ctx, args) => {
    const lead = await ctx.db.get(args.leadId);
    if (!lead) throw new Error("Lead not found");

    const now = Date.now();
    const existingEnriched = lead.enrichedData || {};

    // Merge: mantener datos existentes, agregar LinkedIn
    const mergedData = {
      ...existingEnriched,
      linkedin: args.linkedinData,
      linkedin_enriched: true,
      linkedin_enriched_at: new Date(now).toISOString(),
    };

    await ctx.db.patch(args.leadId, {
      enrichedData: mergedData,
      linkedinUrl: args.linkedinData.linkedinUrl || lead.linkedinUrl,
      lastUpdatedAt: now,
      // Si se detecto tamano de empresa en LinkedIn, actualizar
      ...(args.linkedinData.employeeCount && !lead.tamano ? { tamano: args.linkedinData.employeeCount } : {}),
    });

    return args.leadId;
  },
});
```

```javascript
const lead = $input.item.json;
const scrapeResult = $input.item.json.scrapeData || {};
const markdown = scrapeResult?.data?.markdown || '';
const metadata = scrapeResult?.data?.metadata || {};

if (!markdown || markdown.length < 50) {
  // Scrape failed or empty — skip this lead
  return { json: { leadId: lead._id, success: false, linkedinData: null } };
}

const linkedinData = {
  linkedinUrl: lead.linkedinUrl,
  linkedin_enriched: true,

  // Extraer company size de patrones comunes en LinkedIn
  companySize: null,
  employeeCount: null,
  industry: null,
  recentPosts: [],
  keyContacts: [],
  headquarters: null,
  founded: null,
  specialties: [],
  about: '',
};

// Patterns de LinkedIn company pages
const sizeMatch = markdown.match(/(\d[\d,.]+)\s*(?:employees?|empleados?|seguidores)/i);
if (sizeMatch) {
  const num = sizeMatch[1].replace(/[,.]/g, '');
  linkedinData.employeeCount = num;
  if (parseInt(num) >= 500) linkedinData.companySize = "500+";
  else if (parseInt(num) >= 100) linkedinData.companySize = "100-500";
  else if (parseInt(num) >= 50) linkedinData.companySize = "50-100";
  else linkedinData.companySize = "<50";
}

// Industry
const industryMatch = markdown.match(/(?:industry|industria|sector)[:\s]+([^\n]+)/i);
if (industryMatch) linkedinData.industry = industryMatch[1].trim().substring(0, 100);

// About/description
const aboutMatch = markdown.match(/(?:about|acerca de|descripci[oó]n)[:\s]+([^\n]{20,})/i);
if (aboutMatch) linkedinData.about = aboutMatch[1].trim().substring(0, 500);

// Specialties
const specMatch = markdown.match(/(?:specialties|especialidades)[:\s]+([^\n]+)/i);
if (specMatch) linkedinData.specialties = specMatch[1].split(/[,;]/).map(s => s.trim()).filter(Boolean).slice(0, 10);

// Headquarters
const hqMatch = markdown.match(/(?:headquarters|sede|ubicaci[oó]n)[:\s]+([^\n]+)/i);
if (hqMatch) linkedinData.headquarters = hqMatch[1].trim().substring(0, 100);

// Founded year
const foundedMatch = markdown.match(/(?:founded|fundad[ao]|a[nñ]o)[:\s]+(\d{4})/i);
if (foundedMatch) linkedinData.founded = foundedMatch[1];

return { json: { leadId: lead._id, success: true, linkedinData } };
```
