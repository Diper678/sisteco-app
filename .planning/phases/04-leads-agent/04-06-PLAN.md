1. **moveLeadStage** (mutation): Mueve un lead a un nuevo stage del pipeline.

```typescript
export const moveLeadStage = mutation({
  args: {
    leadId: v.id("leads"),
    newStage: v.string(),
    notes: v.optional(v.string()),
  },
  handler: async (ctx, args) => {
    const lead = await ctx.db.get(args.leadId);
    if (!lead) throw new Error("Lead not found");

    const validStages = ["new", "qualified", "contacted", "meeting", "proposal", "negotiation", "won", "lost"];
    if (!validStages.includes(args.newStage)) {
      throw new Error(`Invalid stage: ${args.newStage}. Valid: ${validStages.join(", ")}`);
    }

    const now = Date.now();
    const history = lead.stageHistory || [];

    // Cerrar stage anterior si existe
    if (history.length > 0 && !history[history.length - 1].exitedAt) {
      history[history.length - 1].exitedAt = now;
    }

    // Agregar nuevo stage
    history.push({
      stage: args.newStage,
      enteredAt: now,
      notes: args.notes,
    });

    await ctx.db.patch(args.leadId, {
      pipelineStage: args.newStage,
      stageHistory: history,
      lastUpdatedAt: now,
    });

    return args.leadId;
  },
});
```

2. **getPipelineStats** (query): Retorna conteo de leads por pipeline stage.

```typescript
export const getPipelineStats = query({
  handler: async (ctx) => {
    const stages = ["new", "qualified", "contacted", "meeting", "proposal", "negotiation", "won", "lost"];
    const stats: Record<string, number> = {};
    let total = 0;

    for (const stage of stages) {
      const leads = await ctx.db
        .query("leads")
        .withIndex("by_pipeline_stage", (q) => q.eq("pipelineStage", stage))
        .collect();
      stats[stage] = leads.length;
      total += leads.length;
    }

    // Leads sin pipeline stage (legacy, pre-pipeline)
    const allLeads = await ctx.db.query("leads").collect();
    const unassigned = allLeads.filter(l => !l.pipelineStage).length;

    return { ...stats, unassigned, total: allLeads.length };
  },
});
```

3. **getLeadsByStage** (query): Retorna leads en un stage especifico.

```typescript
export const getLeadsByStage = query({
  args: {
    stage: v.string(),
    limit: v.optional(v.number()),
  },
  handler: async (ctx, args) => {
    return await ctx.db
      .query("leads")
      .withIndex("by_pipeline_stage", (q) => q.eq("pipelineStage", args.stage))
      .take(args.limit ?? 20);
  },
});
```

4. **autoQualifyHotLeads** (mutation): Busca leads HOT sin pipeline stage o con stage "new" y los mueve a "qualified" automaticamente.

```typescript
export const autoQualifyHotLeads = mutation({
  handler: async (ctx) => {
    const hotLeads = await ctx.db
      .query("leads")
      .withIndex("by_score_category", (q) => q.eq("scoreCategory", "HOT"))
      .collect();

    const now = Date.now();
    let qualified = 0;

    for (const lead of hotLeads) {
      if (!lead.pipelineStage || lead.pipelineStage === "new") {
        const history = lead.stageHistory || [];
        if (history.length > 0 && !history[history.length - 1].exitedAt) {
          history[history.length - 1].exitedAt = now;
        }
        history.push({ stage: "qualified", enteredAt: now, notes: "Auto-qualified by scoring (HOT)" });

        await ctx.db.patch(lead._id, {
          pipelineStage: "qualified",
          stageHistory: history,
          lastUpdatedAt: now,
        });
        qualified++;
      }
    }

    return { qualified, totalHot: hotLeads.length };
  },
});
```
