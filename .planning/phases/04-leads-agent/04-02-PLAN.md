```javascript
const results = $input.all();
const leads = [];

for (const item of results) {
  const data = item.json.data || [];
  for (const result of data) {
    // Extraer empresa del titulo/URL
    const url = result.url || '';
    const domain = url ? new URL(url).hostname.replace('www.', '') : '';
    const empresa = result.title || domain;

    if (!empresa || empresa.length < 3) continue;

    leads.push({
      empresa: empresa.substring(0, 200),
      websiteUrl: url,
      source: "firecrawl_search",
      sourceUrl: url,
      ubicacion: "Chile",
      // Extraer industria del snippet si es posible
      industria: null,
      score: 0,
      scoreCategory: "SKIP",
      status: "new",
    });
  }
}

return [{ json: { leads: leads.slice(0, 50) } }];
```

**Nodo 7 — Parse Enrichment:**
Code node (JavaScript):
```javascript
const lead = $input.item.json.lead;
const scrapeResult = $input.item.json.scrapeData;
const markdown = scrapeResult?.data?.markdown || '';

// Extraer datos del contenido scrapeado
const enrichedData = {
  descripcion: (scrapeResult?.data?.metadata?.description || '').substring(0, 500),
  titulo_pagina: (scrapeResult?.data?.metadata?.title || '').substring(0, 200),
  contenido_preview: markdown.substring(0, 1000),
  // Detectar productos/servicios mencionados
  productos: markdown.match(/(?:productos?|servicios?|soluciones?)\s*[:\-]\s*([^\n]+)/gi)?.slice(0, 5) || [],
  // Detectar stack tecnologico mencionado
  stack_detectado: [],
  // Detectar tamano si se menciona
  tamano_detectado: null,
  industria_detectada: null,
  scraped_at: new Date().toISOString(),
};

// Buscar senales de stack en el HTML/markdown
const stackKeywords = ['hubspot', 'salesforce', 'sap', 'oracle', 'aws', 'azure', 'google cloud', 'shopify', 'magento', 'wordpress'];
for (const kw of stackKeywords) {
  if (markdown.toLowerCase().includes(kw)) {
    enrichedData.stack_detectado.push(kw);
  }
}

return { json: { leadId: lead._id, enrichedData } };
```
